## 1. Setup & Context

### 1.1. Defining a Context
The Context shares data between states. Inherit from `ReactiveContext` for property change notifications.

```csharp
using StateMachine;
using UnityEngine;

public class MyEntityContext : ReactiveContext<MyEntityContext>, IHasAnimator
{
    // References (assigned in StateMachine)
    public Rigidbody2D Rigidbody;
    // Implement IHasAnimator so ConfigurableState can play state animations automatically (no reflection).
    public Animator Animator { get; set; }
    
    // Reactive Properties
    private float moveSpeed;
    public float MoveSpeed 
    { 
        get => moveSpeed; 
        set => SetProperty(ref moveSpeed, value, nameof(MoveSpeed)); 
    }
}
```

### 1.2. The State Machine Controller
The main component that runs the system.

```csharp
using StateMachine;
using UnityEngine;

public class MyEntityStateMachine : StateMachine<MyEntityContext>
{
    [SerializeField] private MyEntityDefinition definition; // Optional: For data-driven setup

    private void Awake()
    {
        // 1. Initialize Context
        // Create context and assign references...
        var context = new MyEntityContext { 
            Rigidbody = GetComponent<Rigidbody2D>(), 
            Animator = GetComponent<Animator>() 
        };
        Initialize(context);

        // 2a. CLASSIC: Register states manually
        // RegisterState("idle", new IdleState());
        
        // 2b. DATA-DRIVEN: Load from Definition (Recommended)
        if (definition != null)
        {
            // Register all states in the definition
            foreach (var config in definition.States)
            {
                if (config != null)
                {
                    RegisterState(config.StateName, config.CreateState());
                }
            }

            // Set Initial State
            if (definition.InitialState != null)
            {
                SetInitialState(definition.InitialState.StateName);
            }
        }
    }
}
```

## 2. Classic Workflow (Code-Only)

### 2.1. Simple State
Inherit from `StateBase`.

```csharp
using StateMachine;
using UnityEngine;

public class BasicMoveState : StateBase<MyEntityContext>
{
    public override void Enter()
    {
        Debug.Log("Started Moving");
        context.Animator.Play("Run");
    }

    public override void Update()
    {
        // Simple movement logic
        Vector2 input = new Vector2(Input.GetAxis("Horizontal"), 0);
        context.Rigidbody.linearVelocity = input * context.MoveSpeed;
        
        // Manual transition check
        if (input.magnitude < 0.01f)
        {
            stateMachine.ChangeState("Idle");
        }
    }
}
```

### 2.2. Builder Pattern (No Classes)
Great for prototyping or simple logic.

```csharp
var jumpState = new StateBuilder<MyEntityContext>()
    .SetName("Jump")
    .OnEnter(() => {
        context.Rigidbody.AddForce(Vector2.up * 10, ForceMode2D.Impulse);
    })
    .Build();

stateMachine.RegisterState("Jump", jumpState);
```

## 3. Data-Driven Workflow (Recommended)

### 3.1. Create a Config
This holds the data. Usually generated by the **State Machine Creator** tool.

```csharp
using StateMachine;
using UnityEngine;

[CreateAssetMenu(menuName = "MyEnt/Configs/Attack")]
public class AttackConfig : StateConfig<MyEntityContext>, IStateConfigWithAnimation
{
    [Header("Settings")]
    public float damage = 10f;
    public float duration = 1.5f;
    
    [Header("Visuals")]
    [SerializeField] private AnimationClip animation;
    
    // IStateConfigWithAnimation implementation
    public AnimationClip Animation => animation;
    public string AnimatorStateName => null; // Use state name
    public string AnimatorLayerName => "Base Layer";

    // Factory method
    public override IState<MyEntityContext> CreateState() => new AttackState(this);
}
```

### 3.2. Create the Configurable State
This holds the logic. It receives the specific config type.

```csharp
using StateMachine;
using UnityEngine;

// <Context, ConfigType, StateMachineType>
public class AttackState : ConfigurableState<MyEntityContext, AttackConfig, MyEntityStateMachine>
{
    private float timer;

    public AttackState(AttackConfig config) : base(config) { }

    public override void Enter()
    {
        // 1. Base plays the config's animation automatically — the config implements
        //    IStateConfigWithAnimation, and the context implements IHasAnimator (so the Animator
        //    is resolved without reflection).
        base.Enter(); 
        
        // 2. Initialize logic
        timer = 0;
        Debug.Log($"Attacking with {config.damage} damage!");
    }

    public override void Update()
    {
        base.Update(); // Updates transitions
        
        timer += Time.deltaTime;
        if (timer >= config.duration)
        {
            stateMachine.ChangeState("Idle");
        }
    }
}
```

### 3.3. Custom Conditions
Create a new rule for transitions by inheriting from `TransitionCondition<TContext>`.

```csharp
using StateMachine.Core;
using UnityEngine;

[CreateAssetMenu(menuName = "Conditions/IsGrounded")]
public class IsGroundedCondition : TransitionCondition<MyEntityContext>
{
    public override bool Check(MyEntityContext context)
    {
        // Assumes context has an IsGrounded property
        return context.IsGrounded;
    }
}
```

## 4. Advanced

### 4.1. Listening to State Changes

```csharp
private void OnEnable()
{
    stateMachine.OnStateChanged += HandleStateChange;
}

private void HandleStateChange(IState<MyEntityContext> oldState, IState<MyEntityContext> newState)
{
    Debug.Log($"Transformed from {oldState?.GetType().Name} to {newState.GetType().Name}");
}
```

### 4.2. Reactive Properties
Reacting to context data changes.

```csharp
// Inside your Context class
public int Health 
{ 
    get => health; 
    set => SetProperty(ref health, value, nameof(Health)); 
}

// Inside any script (UI, etc.)
context.OnPropertyChanged(nameof(MyEntityContext.Health), (oldVal, newVal) => {
    UpdateHealthBar(newVal);
});
```
